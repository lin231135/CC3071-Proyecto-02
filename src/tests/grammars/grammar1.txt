# Gramática 1 — Expresiones aritméticas (Libro del Dragón §4.28)
# Complejidad: media
#
# PRIMERO esperado:
#   E = { ID, LPAREN }
#   T = { ID, LPAREN }
#   F = { ID, LPAREN }
#
# SIGUIENTE esperado:
#   E = { $, PLUS, RPAREN }
#   T = { $, PLUS, RPAREN, TIMES }
#   F = { $, PLUS, RPAREN, TIMES }

E -> E PLUS T | T
T -> T TIMES F | F
F -> LPAREN E RPAREN | ID
